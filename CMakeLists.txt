cmake_minimum_required(VERSION 3.16)
project(mlu_operator C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Werror -ffp-contract=off)
enable_testing()
add_executable(gemm_reference operators/gemm/tests/reference_test.c)
target_include_directories(gemm_reference PRIVATE common/reference)
target_link_libraries(gemm_reference PRIVATE m)
add_test(NAME gemm_reference COMMAND gemm_reference)

option(MLU_NATIVE "Build CNNL operator objects" OFF)
option(MLU_LINK "Link tests against the installed SDK" OFF)
set(NEUWARE_ROOT "/usr/local/neuware" CACHE PATH "Cambricon SDK prefix")
if(MLU_LINK AND NOT MLU_NATIVE)
    message(FATAL_ERROR "MLU_LINK requires MLU_NATIVE")
endif()
if(MLU_NATIVE)
    add_library(cnrt_probe_object OBJECT common/runtime/cnrt_probe.c)
    target_include_directories(cnrt_probe_object PRIVATE "${NEUWARE_ROOT}/include")
    if(MLU_LINK)
        add_executable(cnrt_probe $<TARGET_OBJECTS:cnrt_probe_object>)
        target_link_directories(cnrt_probe PRIVATE "${NEUWARE_ROOT}/lib64")
        target_link_options(cnrt_probe PRIVATE "-Wl,-rpath-link,${NEUWARE_ROOT}/lib64")
        target_link_libraries(cnrt_probe PRIVATE cnrt)
    endif()
    foreach(op IN ITEMS gemm rmsnorm)
        add_library(${op}_operator STATIC operators/${op}/src/mlu_${op}.c)
        target_include_directories(${op}_operator PUBLIC operators/${op}/include common/runtime
                                   PRIVATE "${NEUWARE_ROOT}/include")
        if(MLU_LINK)
            add_executable(${op}_native_test operators/${op}/tests/native_test.c)
            target_include_directories(${op}_native_test PRIVATE common/reference)
            target_link_directories(${op}_native_test PRIVATE "${NEUWARE_ROOT}/lib64")
            target_link_options(${op}_native_test PRIVATE "-Wl,-rpath-link,${NEUWARE_ROOT}/lib64")
            target_link_libraries(${op}_native_test PRIVATE ${op}_operator cnnl cnrt m)
        else()
            add_library(${op}_test_object OBJECT operators/${op}/tests/native_test.c)
            target_include_directories(${op}_test_object PRIVATE common/reference)
            target_link_libraries(${op}_test_object PRIVATE ${op}_operator)
        endif()
    endforeach()
endif()
